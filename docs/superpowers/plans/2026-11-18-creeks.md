## Data shape (shared by Python and TypeScript)

`creeks.json`:

```json
{
  "version": 1,
  "mapped": [
    {"id": "creek-9", "name": "Chutchui Creek", "confidence": "inferred",
     "sources": "Hoadley 1855; Horner 1854; Goddard 1869; Seep City", "points": [x0, z0, x1, z1]}
  ],
  "modelled": [{"order": 4, "points": [x0, z0, x1, z1]}],
  "processing": {"tolerance_m": 3.0, "min_order": 3}
}
```

`points` is a flat `[x, z, ...]` array in viewer world metres (x east, z south, origin at the terrain centre), rounded to 0.1 m. `name` is `null` for unnamed lines. Confidence is `documented` (SFEI marks the line "well located"), `inferred` (other hand-mapped lines) or `illustrative` (modelled channels are not stored with a confidence; the meta row carries `illustrative`).

Limits, mirrored in `parse.ts` and `creeks.py` (a test compares them): `MAX_MAPPED = 500`, `MAX_MODELLED = 20_000`, `MAX_CREEK_POINTS = 400_000`, `MAX_NAME_CHARS = 120`, `MAX_SOURCE_CHARS = 300`, `MAX_COORD_M = 1e6`, orders 1 to 8.

`meta.json` additions: `creeks_file: "creeks.json"`, `creeks_mapped: <count>`, `creeks_modelled: <count>`, and two `hand_sources` rows: layer `creeks` and layer `modelled_streams`.

```python
"""Historical creeks: SFEI's hand-mapped creek lines and its modelled stream channels, in the
viewer's WORLD coordinates (see overlay.py: metres from the terrain centre, x east, z south).

Two public SFEI ArcGIS layers feed it (neither states a licence; see
docs/superpowers/specs/2026-11-18-creeks-design.md):

* "San Francisco Historical Creeks": 26 hand-mapped lines for the north and centre of the city,
  with names, a "well located" flag and cited sources.
* "MYHL_HistoricStreams": stream channels computed from a historical elevation model for the
  south. Only Strahler order 3 and up are kept, merged end to end into channels.

The result is written as `creeks.json` next to `heightmap.u16`, and `meta.json` gains
`creeks_file`, `creeks_mapped`, `creeks_modelled` and two `hand_sources` rows. A city rebuild
rewrites meta.json, so run build_creeks.py again afterwards.

    uv run python scripts/build_creeks.py ../viewer/public/assets/real-city
"""

from __future__ import annotations

import json
import math
import re
from collections import defaultdict
from pathlib import Path
from typing import Any, Sequence

import numpy as np
from shapely.geometry import LineString

from yelamu.export import CONFIDENCE_LEVELS
from yelamu.geo import Grid
from yelamu.overlay import lonlat_to_world
from yelamu.provenance import lowest_confidence

CREEKS_VERSION = 1
CREEKS_FILE = "creeks.json"

# Limits of the viewer's creeks parser. The viewer rejects the WHOLE file when one is exceeded,
# so the validator below must be at least as strict.
# keep in sync with viewer/src/creeks/parse.ts
MAX_MAPPED = 500
MAX_MODELLED = 20_000
MAX_CREEK_POINTS = 400_000
MAX_NAME_CHARS = 120
MAX_SOURCE_CHARS = 300
MAX_COORD_M = 1e6
MIN_ORDER, MAX_ORDER = 1, 8

MIN_MODELLED_ORDER = 3
SIMPLIFY_TOLERANCE_M = 3.0
# A dry run shorter than this after trimming at the bay is not worth drawing.
MIN_RUN_LENGTH_M = 30.0
# Segments of the modelled network are joined when their end points agree to this many metres.
JOIN_KEY_M = 0.1

# Habitat classes (meta.habitat_classes names) where a creek line must stop: open bay and ocean.
# Ponds, lagoons, marsh and tidal flat are NOT here: creeks run into those.
BAY_CLASS_NAMES = frozenset({
    "Open bay (inferred from modern elevation)",
    "Bay (hand-drawn)",
    "Deep ocean/bay",
    "Shallow ocean/bay",
})

XZ = tuple[float, float]


# ----------------------------------------------------------------------------- reading


def read_geojson_lines(path: Path | str) -> list[dict]:
    """[{'props': {...}, 'lines': [[(lon, lat), ...], ...]}] for a GeoJSON file of lines."""
    path = Path(path)
    if not path.exists():
        raise FileNotFoundError(f"creek file not found: {path}")
    try:
        doc = json.loads(path.read_text())
    except json.JSONDecodeError as exc:
        raise ValueError(f"{path}: not valid JSON ({exc})") from exc
    if not isinstance(doc, dict) or doc.get("type") != "FeatureCollection":
        raise ValueError(f"{path}: expected a GeoJSON FeatureCollection")
    out: list[dict] = []
    for i, feature in enumerate(doc.get("features") or []):
        geometry = (feature or {}).get("geometry") or {}
        kind = geometry.get("type")
        if kind == "LineString":
            lines = [geometry["coordinates"]]
        elif kind == "MultiLineString":
            lines = geometry["coordinates"]
        else:
            raise ValueError(f"{path}: feature {i}: expected a line geometry, got {kind!r}")
        out.append({
            "props": feature.get("properties") or {},
            "lines": [[(float(c[0]), float(c[1])) for c in line] for line in lines],
        })
    return out


def project_line(grid: Grid, lonlat: Sequence[tuple[float, float]]) -> list[XZ]:
    """Longitude and latitude to viewer world metres."""
    return [lonlat_to_world(grid, lon, lat) for lon, lat in lonlat]


def mapped_confidence(props: dict) -> str:
    """SFEI's own flag: 'well located' lines are documented, every other hand-mapped line is inferred."""
    flag = str(props.get("stream_typ") or "").strip().lower()
    return "documented" if flag == "well located" else "inferred"


# ------------------------------------------------------------------ trimming, simplifying


def bay_mask(habitat: np.ndarray, classes: Sequence[dict]) -> np.ndarray:
    """Boolean (rows, cols) mask of the open bay and ocean cells of a habitat raster."""
    ids = [int(c["id"]) for c in classes if c.get("name") in BAY_CLASS_NAMES]
    return np.isin(habitat, ids)


def _length(points: Sequence[XZ]) -> float:
    return sum(math.hypot(b[0] - a[0], b[1] - a[1]) for a, b in zip(points, points[1:]))


def trim_to_land(
    points: Sequence[XZ],
    bay: np.ndarray,
    grid_w: int,
    grid_h: int,
    cell: float,
    *,
    min_length_m: float = MIN_RUN_LENGTH_M,
) -> list[list[XZ]]:
    """The dry runs of a polyline: pieces whose points are not in the bay mask (a point outside the
    grid counts as wet). Pieces with fewer than two points or shorter than `min_length_m` are dropped.
    The grid cell of world (x, z) is col = x / cell + (w - 1) / 2, row = z / cell + (h - 1) / 2, the
    same frame as overlay.py and the viewer's terrain mesh."""
    runs: list[list[XZ]] = []
    current: list[XZ] = []
    for x, z in points:
        col = math.floor(x / cell + (grid_w - 1) / 2 + 0.5)
        row = math.floor(z / cell + (grid_h - 1) / 2 + 0.5)
        wet = not (0 <= col < grid_w and 0 <= row < grid_h) or bool(bay[row, col])
        if wet:
            if current:
                runs.append(current)
                current = []
        else:
            current.append((x, z))
    if current:
        runs.append(current)
    return [r for r in runs if len(r) >= 2 and _length(r) >= min_length_m]


def simplify_line(points: Sequence[XZ], tolerance_m: float) -> list[XZ]:
    """Douglas-Peucker; the end points are always kept."""
    if len(points) < 3:
        return [(float(x), float(z)) for x, z in points]
    simple = LineString(points).simplify(tolerance_m, preserve_topology=False)
    return [(float(x), float(z)) for x, z in simple.coords]


# ----------------------------------------------------------------------------- chaining


def _key(p: XZ) -> tuple[int, int]:
    return (round(p[0] / JOIN_KEY_M), round(p[1] / JOIN_KEY_M))


def chain_segments(segments: Sequence[tuple[int, Sequence[XZ]]]) -> list[tuple[int, list[XZ]]]:
    """Merge segments into channels. Two segments are joined where exactly two segment ends meet
    at a point and both have the same Strahler order. A confluence (three or more ends) or a
    change of order keeps the segments separate. The input is not modified. The result is sorted
    by (order, first point), so it does not depend on the input order."""
    segs: dict[int, tuple[int, list[XZ]]] = {
        i: (order, [(float(x), float(z)) for x, z in pts]) for i, (order, pts) in enumerate(segments)
    }
    while True:
        ends: dict[tuple[int, int], list[tuple[int, int]]] = defaultdict(list)
        for i, (_, pts) in segs.items():
            ends[_key(pts[0])].append((i, 0))
            ends[_key(pts[-1])].append((i, 1))
        touched: set[int] = set()
        merged = False
        for node in sorted(ends):
            items = ends[node]
            if len(items) != 2:
                continue
            (a, end_a), (b, end_b) = items
            if a == b or a in touched or b in touched or segs[a][0] != segs[b][0]:
                continue
            pa = list(segs[a][1])
            pb = list(segs[b][1])
            if end_a == 0:
                pa.reverse()  # make `a` finish at the node
            if end_b == 1:
                pb.reverse()  # make `b` start at the node
            segs[a] = (segs[a][0], pa + pb[1:])
            del segs[b]
            touched.update((a, b))
            merged = True
        if not merged:
            break
    return sorted(segs.values(), key=lambda s: (s[0], s[1][0], s[1][-1]))
```

```python
def _lonlat_line(grid, x0, z0, x1, z1):
```

```python
MAPPED_LAYER = "creeks"
MODELLED_LAYER = "modelled_streams"
# Proposed wording for the Sources panel; the owner approves it verbatim before it ships.
MAPPED_SOURCE_NOTE = (
    "Creek lines from SFEI's 'San Francisco Historical Creeks' layer (Hidden Nature SF), mapped from "
    "1850s maps and a historical elevation model. Approximate."
)
MODELLED_SOURCE_NOTE = (
    "Stream channels computed by SFEI from a historical elevation model ('MYHL_HistoricStreams'), "
    "Strahler order 3 and up. Computed, not observed."

def _flat(points: Sequence[XZ]) -> list[float]:
    out: list[float] = []
    for x, z in points:
        out.append(round(float(x), 1) + 0.0)
        out.append(round(float(z), 1) + 0.0)
    return out


def _clip(text: str, limit: int) -> str:
    return text if len(text) <= limit else text[: limit - 1] + "…"


def read_modelled_lines(path: Path | str) -> list[dict]:
    """[{'order': int, 'line': [(lon, lat), ...]}] from the modelled-streams GeoJSON (field StrahlerID)."""
    out: list[dict] = []
    for i, feature in enumerate(read_geojson_lines(path)):
        order = feature["props"].get("StrahlerID")
        if not isinstance(order, (int, float)) or isinstance(order, bool) or int(order) != order:
            raise ValueError(f"{path}: feature {i}: StrahlerID must be a whole number, got {order!r}")
        for line in feature["lines"]:
            out.append({"order": int(order), "line": line})
    return out


def build_creeks_document(
    grid: Grid,
    mapped: Sequence[dict],
    modelled: Sequence[dict],
    bay: np.ndarray,
    *,
    min_order: int = MIN_MODELLED_ORDER,
    tolerance_m: float = SIMPLIFY_TOLERANCE_M,
) -> tuple[dict, dict]:
    """(creeks.json document, statistics). `mapped` is read_geojson_lines output, `modelled` is
    read_modelled_lines output, `bay` the bay/ocean mask on the city grid."""
    rows, cols = bay.shape

    def runs_for(lonlat: Sequence[tuple[float, float]]) -> list[list[XZ]]:
        return [simplify_line(run, tolerance_m)
                for run in trim_to_land(project_line(grid, lonlat), bay, cols, rows, grid.cell)]

    mapped_out: list[dict] = []
    for feature in sorted(mapped, key=lambda f: int(f["props"].get("OBJECTID", 0))):
        props = feature["props"]
        oid = int(props.get("OBJECTID", 0))
        name = str(props.get("NAME1") or "").strip() or None
        sources = _clip(str(props.get("S_Interp1") or "").strip(), MAX_SOURCE_CHARS)
        k = 0
        for lonlat in feature["lines"]:
            for run in runs_for(lonlat):
                k += 1
                mapped_out.append({
                    "id": f"creek-{oid}" if k == 1 else f"creek-{oid}-{k}",
                    "name": name,
                    "confidence": mapped_confidence(props),
                    "sources": sources,
                    "points": _flat(run),
                })

    kept = [m for m in modelled if m["order"] >= min_order]
    segments: list[tuple[int, list[XZ]]] = []
    for m in kept:
        for run in trim_to_land(project_line(grid, m["line"]), bay, cols, rows, grid.cell, min_length_m=0.0):
            segments.append((m["order"], run))
    channels = chain_segments(segments)
    modelled_out: list[dict] = []
    for order, pts in channels:
        simple = simplify_line(pts, tolerance_m)
        if len(simple) >= 2 and _length(simple) >= MIN_RUN_LENGTH_M:
            modelled_out.append({"order": order, "points": _flat(simple)})

    doc = {
        "version": CREEKS_VERSION,
        "mapped": mapped_out,
        "modelled": modelled_out,
        "processing": {"tolerance_m": float(tolerance_m), "min_order": int(min_order)},
    }
    stats = {
        "mapped": len(mapped_out),
        "modelled": len(modelled_out),
        "modelled_segments": len(kept),
        "dropped_low_order": len(modelled) - len(kept),
    }
    return doc, stats

_ID_OK = re.compile(r"[a-z0-9][a-z0-9-]*")


def _check_points(points: Any, where: str, budget: list[int]) -> None:
    if not isinstance(points, list) or len(points) < 4 or len(points) % 2 != 0:
        raise ValueError(f"{where}: points must be a flat [x, z, ...] list with at least 2 points")
    budget[0] += len(points) // 2
    if budget[0] > MAX_CREEK_POINTS:
        raise ValueError(f"creeks hold more than {MAX_CREEK_POINTS} points in total")
    for k, v in enumerate(points):
        if isinstance(v, bool) or not isinstance(v, (int, float)) or not math.isfinite(v):
            raise ValueError(f"{where}: value {k} is not a finite number")
        if abs(v) > MAX_COORD_M:
            raise ValueError(f"{where}: value {k} is out of range (within +-1e6 metres), got {v}")


def validate_creeks_document(doc: Any) -> None:
    """Raise ValueError for anything the viewer's parser (viewer/src/creeks/parse.ts) would reject."""
    if not isinstance(doc, dict) or doc.get("version") != CREEKS_VERSION:
        raise ValueError(f"creeks: unsupported version {None if not isinstance(doc, dict) else doc.get('version')!r}")
    mapped, modelled = doc.get("mapped"), doc.get("modelled")
    if not isinstance(mapped, list) or len(mapped) > MAX_MAPPED:
        raise ValueError(f"creeks: mapped must be a list of at most {MAX_MAPPED}")
    if not isinstance(modelled, list) or len(modelled) > MAX_MODELLED:
        raise ValueError(f"creeks: modelled must be a list of at most {MAX_MODELLED}")
    budget = [0]
    seen: set[str] = set()
    for i, c in enumerate(mapped):
        where = f"mapped[{i}]"
        cid = c.get("id")
        if not isinstance(cid, str) or not _ID_OK.fullmatch(cid):
            raise ValueError(f"{where}: id {cid!r} must match [a-z0-9][a-z0-9-]*")
        if cid in seen:
            raise ValueError(f"{where}: duplicate id {cid!r}")
        seen.add(cid)
        name = c.get("name")
        if name is not None and (not isinstance(name, str) or not name.strip() or len(name) > MAX_NAME_CHARS):
            raise ValueError(f"{where} ({cid}): name must be null or 1 to {MAX_NAME_CHARS} characters")
        if c.get("confidence") not in CONFIDENCE_LEVELS:
            raise ValueError(f"{where} ({cid}): confidence {c.get('confidence')!r} is not one of {CONFIDENCE_LEVELS}")
        sources = c.get("sources")
        if not isinstance(sources, str) or len(sources) > MAX_SOURCE_CHARS:
            raise ValueError(f"{where} ({cid}): sources must be text of at most {MAX_SOURCE_CHARS} characters")
        _check_points(c.get("points"), f"{where} ({cid})", budget)
    for i, m in enumerate(modelled):
        where = f"modelled[{i}]"
        order = m.get("order")
        if isinstance(order, bool) or not isinstance(order, int) or not MIN_ORDER <= order <= MAX_ORDER:
            raise ValueError(f"{where}: order must be a whole number from {MIN_ORDER} to {MAX_ORDER}")
        _check_points(m.get("points"), where, budget)


# --------------------------------------------------------------------------- meta patch


def patch_creeks_meta(meta: dict, doc: dict | None) -> dict:
    """The meta.json dict with the creek keys set; returns a NEW dict, `meta` is untouched.

    Sets `creeks_file`, `creeks_mapped`, `creeks_modelled` and one `hand_sources` row per layer
    that has lines (so the Sources panel lists it). With `doc` None, or no lines at all, every
    creek key and row is removed. Other keys and other layers' rows are kept, in order. A city
    rebuild rewrites meta.json, so build_creeks has to be run again afterwards."""
    out = {k: v for k, v in meta.items() if not k.startswith("creeks_")}
    rows = [r for r in meta.get("hand_sources", []) if r.get("layer") not in (MAPPED_LAYER, MODELLED_LAYER)]
    mapped = (doc or {}).get("mapped", [])
    modelled = (doc or {}).get("modelled", [])
    if mapped or modelled:
        out["creeks_file"] = CREEKS_FILE
        out["creeks_mapped"] = len(mapped)
        out["creeks_modelled"] = len(modelled)
    if mapped:
        rows.append({"layer": MAPPED_LAYER, "source": MAPPED_SOURCE_NOTE,
                     "confidence": lowest_confidence(c["confidence"] for c in mapped), "count": len(mapped)})
    if modelled:
        rows.append({"layer": MODELLED_LAYER, "source": MODELLED_SOURCE_NOTE,
                     "confidence": "illustrative", "count": len(modelled)})
    if rows:
        out["hand_sources"] = rows
    else:
        out.pop("hand_sources", None)
    return out

def build_creeks(
    assets: Path | str,
    mapped_path: Path | str | None,
    modelled_path: Path | str | None,
    *,
    remove: bool = False,
    min_order: int = MIN_MODELLED_ORDER,
    tolerance_m: float = SIMPLIFY_TOLERANCE_M,
) -> dict | None:
    """Write creeks.json into `assets` (the folder with the city's meta.json and habitat raster)
    and patch meta.json. `remove=True` deletes creeks.json and every creek key instead. Nothing is
    written if an input is invalid. Re-running is idempotent. Returns the document (None on remove)."""
    assets = Path(assets)
    meta_path = assets / "meta.json"
    if not meta_path.exists():
        raise FileNotFoundError(f"{meta_path} not found: build the terrain assets into {assets} first")
    meta = json.loads(meta_path.read_text())

    if remove:
        patched = patch_creeks_meta(meta, None)
        (assets / CREEKS_FILE).unlink(missing_ok=True)
        if patched != meta:
            meta_path.write_text(json.dumps(patched, indent=2))
        return None

    grid = Grid(meta["west"], meta["south"], meta["east"], meta["north"], float(meta["cell_m"]))
    if (grid.width, grid.height) != (meta["width"], meta["height"]):
        raise ValueError(f"{meta_path}: width and height do not match the grid built from its bounds")
    habitat = np.frombuffer((assets / meta["habitat_file"]).read_bytes(), dtype=np.uint8)
    if habitat.size != grid.width * grid.height:
        raise ValueError(f"{assets / meta['habitat_file']}: size does not match the {grid.width} x {grid.height} grid")
    bay = bay_mask(habitat.reshape(grid.height, grid.width), meta.get("habitat_classes", []))

    mapped = read_geojson_lines(mapped_path) if mapped_path is not None else []
    modelled = read_modelled_lines(modelled_path) if modelled_path is not None else []
    doc, _ = build_creeks_document(grid, mapped, modelled, bay, min_order=min_order, tolerance_m=tolerance_m)
    validate_creeks_document(doc)

    (assets / CREEKS_FILE).write_text(json.dumps(doc, separators=(",", ":")))
    patched = patch_creeks_meta(meta, doc)
    if patched != meta:
        meta_path.write_text(json.dumps(patched, indent=2))
    return doc
```

```python
"""Download SFEI's two public creek layers to pipeline/data/raw/sfei-creeks/ (git-ignored).

    uv run python scripts/fetch_creeks.py mapped      # 26 hand-mapped lines, about 84 KB
    uv run python scripts/fetch_creeks.py modelled    # modelled streams, Strahler order 3 and up
    uv run python scripts/fetch_creeks.py both

Both are public ArcGIS feature services owned by SFEI's Resilient Landscapes programme. Neither
states a licence; see docs/superpowers/specs/2026-11-18-creeks-design.md and NOTICE.md. Only the
query results are saved: nothing is committed.
"""

from __future__ import annotations

import argparse
import json
import sys
import urllib.parse
import urllib.request
from dataclasses import dataclass
from pathlib import Path
from typing import Sequence

BASE = "https://services1.arcgis.com/8V9ShUQ1DGYVLtOg/arcgis/rest/services"
PAGE = 2000  # the services' maxRecordCount


@dataclass(frozen=True)
class Source:
    name: str
    service: str
    layer: int
    where: str
    out_file: str


MAPPED = Source("hand-mapped creeks", "San_Francisco_Historical_Creeks", 0, "1=1",
                "san_francisco_historical_creeks.geojson")
MODELLED = Source("modelled streams, order 3 and up", "MYHL_HistoricStreams", 1, "StrahlerID>=3",
                  "myhl_historic_streams_order3.geojson")


def query_url(source: Source, *, offset: int) -> str:
    params = {
        "where": source.where,
        "outFields": "*",
        "outSR": "4326",
        "orderByFields": "OBJECTID",
        "resultOffset": str(offset),
        "resultRecordCount": str(PAGE),
        "f": "geojson",
    }
    return f"{BASE}/{source.service}/FeatureServer/{source.layer}/query?{urllib.parse.urlencode(params)}"


def fetch(source: Source, dest_dir: Path) -> Path:
    features: list[dict] = []
    offset = 0
    while True:
        with urllib.request.urlopen(query_url(source, offset=offset), timeout=60) as resp:  # noqa: S310
            page = json.load(resp)
        batch = page.get("features", [])
        features.extend(batch)
        if len(batch) < PAGE and not page.get("exceededTransferLimit"):
            break
        offset += PAGE
    dest_dir.mkdir(parents=True, exist_ok=True)
    out = dest_dir / source.out_file
    out.write_text(json.dumps({"type": "FeatureCollection", "features": features}))
    print(f"{source.name}: {len(features)} features, {out.stat().st_size / 1024:.0f} KB -> {out}")
    return out


def main(argv: Sequence[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    parser.add_argument("which", choices=("mapped", "modelled", "both"))
    args = parser.parse_args(argv)
    dest = Path(__file__).resolve().parent.parent / "data" / "raw" / "sfei-creeks"
    try:
        if args.which in ("mapped", "both"):
            fetch(MAPPED, dest)
        if args.which in ("modelled", "both"):
            fetch(MODELLED, dest)
    except OSError as exc:
        print(f"error: {exc}", file=sys.stderr)
        return 1
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```ts
import type { HeightmapMeta } from "../heightmap";
import { CONFIDENCES } from "../overlay/types";
import type { Creeks, MappedCreek, ModelledStream } from "./types";

/**
 * Protective caps: a hostile or broken file must not be able to exhaust memory.
 * KEEP IN SYNC with pipeline/yelamu/creeks.py: the pipeline refuses to write a creeks.json this
 * parser would reject, and a pipeline test reads this file and fails when the numbers differ.
 */
export const MAX_MAPPED = 500;
export const MAX_MODELLED = 20_000;
export const MAX_CREEK_POINTS = 400_000;
export const MAX_NAME_CHARS = 120;
export const MAX_SOURCE_CHARS = 300;
/** World coordinates (metres from the terrain centre) must stay within this. */
export const MAX_COORD_M = 1e6;
export const MIN_ORDER = 1;
export const MAX_ORDER = 8;

const ID = /^[a-z0-9][a-z0-9-]*$/;

function fail(where: string, what: string): never {
  throw new Error(`creeks: ${where}: ${what}`);
}

function isRecord(v: unknown): v is Record<string, unknown> {
  return typeof v === "object" && v !== null && !Array.isArray(v);
}

function parsePoints(v: unknown, label: string, budget: { points: number }): number[] {
  if (!Array.isArray(v)) fail(label, "points must be a flat array of numbers");
  if (v.length % 2 !== 0) fail(label, `points has an odd number of values (${v.length})`);
  if (v.length < 4) fail(label, `needs at least 2 points, has ${v.length / 2}`);
  budget.points += v.length / 2;
  if (budget.points > MAX_CREEK_POINTS) fail(label, `creeks hold more than ${MAX_CREEK_POINTS} points in total`);
  const out = new Array<number>(v.length);
  for (let k = 0; k < v.length; k++) {
    const n: unknown = v[k];
    if (typeof n !== "number" || !Number.isFinite(n)) fail(label, `value ${k} is not a finite number`);
    if (Math.abs(n) > MAX_COORD_M) fail(label, `value ${k} is out of range (must be within ±1e6 metres), got ${n}`);
    out[k] = n;
  }
  return out;
}

/**
 * Validate untrusted creeks JSON and return a clean copy (only the known fields). A name that is
 * only whitespace becomes null. Throws an Error naming the line.
 */
export function parseCreeks(raw: unknown): Creeks {
  if (!isRecord(raw)) throw new Error("creeks: the file must be a JSON object");
  if (raw.version !== 1) {
    throw new Error(`creeks: unsupported version ${JSON.stringify(raw.version)} (expected 1)`);
  }
  const mappedRaw = raw.mapped;
  const modelledRaw = raw.modelled;
  if (!Array.isArray(mappedRaw)) fail("mapped", "must be an array");
  if (!Array.isArray(modelledRaw)) fail("modelled", "must be an array");
  if (mappedRaw.length > MAX_MAPPED) fail("mapped", `has ${mappedRaw.length} entries, more than the limit of ${MAX_MAPPED}`);
  if (modelledRaw.length > MAX_MODELLED) {
    fail("modelled", `has ${modelledRaw.length} entries, more than the limit of ${MAX_MODELLED}`);
  }

  const budget = { points: 0 };
  const seen = new Set<string>();
  const mapped = mappedRaw.map((c: unknown, i: number): MappedCreek => {
    const where = `mapped[${i}]`;
    if (!isRecord(c)) fail(where, "must be an object");
    if (typeof c.id !== "string" || !ID.test(c.id)) fail(where, `id ${JSON.stringify(c.id)} must match ${ID.source}`);
    const label = `${where} (${c.id})`;
    if (seen.has(c.id)) fail(label, `duplicate id "${c.id}"`);
    seen.add(c.id);
    let name: string | null = null;
    if (c.name !== null && c.name !== undefined) {
      if (typeof c.name !== "string") fail(label, "name must be text or null");
      if (c.name.length > MAX_NAME_CHARS) fail(label, `name is longer than ${MAX_NAME_CHARS} characters`);
      name = c.name.trim() === "" ? null : c.name;
    }
    if (typeof c.sources !== "string") fail(label, "sources must be text");
    if (c.sources.length > MAX_SOURCE_CHARS) fail(label, `sources is longer than ${MAX_SOURCE_CHARS} characters`);
    const confidence = c.confidence;
    if (typeof confidence !== "string" || !(CONFIDENCES as readonly string[]).includes(confidence)) {
      fail(label, `confidence ${JSON.stringify(confidence)} is not one of ${CONFIDENCES.join(", ")}`);
    }
    return {
      id: c.id,
      name,
      confidence: confidence as MappedCreek["confidence"],
      sources: c.sources,
      points: parsePoints(c.points, label, budget),
    };
  });
  const modelled = modelledRaw.map((m: unknown, i: number): ModelledStream => {
    const where = `modelled[${i}]`;
    if (!isRecord(m)) fail(where, "must be an object");
    const order = m.order;
    if (typeof order !== "number" || !Number.isInteger(order) || order < MIN_ORDER || order > MAX_ORDER) {
      fail(where, `order ${JSON.stringify(order)} must be a whole number from ${MIN_ORDER} to ${MAX_ORDER}`);
    }
    return { order, points: parsePoints(m.points, where, budget) };
  });
  return { version: 1, mapped, modelled };
}

type FetchLike = (url: string) => Promise<Pick<Response, "ok" | "status" | "json">>;

/**
 * Fetch and validate the creeks file named by `meta.creeks_file` next to the heightmap. Resolves
 * null when the dataset has none. Rejects with the URL (and HTTP status) on a failed request, or
 * with the parser's message on bad content.
 */
export async function loadCreeks(
  baseUrl: string,
  meta: Pick<HeightmapMeta, "creeks_file">,
  fetchImpl: FetchLike = (u) => fetch(u),
): Promise<Creeks | null> {
  const file = meta.creeks_file;
  if (file === undefined || file === null || file === "") return null;
  if (typeof file !== "string" || !/^[A-Za-z0-9][A-Za-z0-9._-]*$/.test(file) || file.includes("..")) {
    throw new Error(`creeks: unsupported creeks_file ${JSON.stringify(file)}`);
  }
  const url = `${baseUrl}/${file}`;
  const res = await fetchImpl(url);
  if (!res.ok) throw new Error(`failed to fetch ${url}: HTTP ${res.status}`);
  let json: unknown;
  try {
    json = await res.json();
  } catch (err) {
    throw new Error(`${url}: not valid JSON (${err instanceof Error ? err.message : String(err)})`);
  }
  try {
    return parseCreeks(json);
  } catch (err) {
    throw new Error(`${url}: ${err instanceof Error ? err.message : String(err)}`);
  }
}
```

```ts
import { drapePolyline, type HeightAt } from "../overlay/drape";

export interface RibbonOptions {
  /** Ground height in metres, NOT exaggerated. */
  heightAt: HeightAt;
  waterLevelM: number;
  /** Horizontal width of the ribbon, metres. */
  widthM: number;
  /** Longest horizontal distance between draped centre-line points, metres. */
  stepM: number;
  /** Metres above max(ground, water), before the exaggeration. */
  liftM: number;
  exaggeration: number;
}

export interface RibbonMesh {
  /** Flat [x, y, z, ...], y already exaggerated. */
  positions: Float32Array;
  indices: Uint32Array;
}

/**
 * A flat ribbon along each polyline, draped over the terrain. Each line is subdivided with
 * drapePolyline (so it follows the relief), then every centre-line point gets a left and a right
 * vertex half a width away, perpendicular to the line there (the direction is averaged over the
 * neighbouring segments, so a bend does not open a gap). Every vertex sits at
 * (max(ground, water) + lift) * exaggeration, so it is above ground and above the sea. Lines with
 * fewer than two distinct points are skipped. Deterministic; triangles are drawn double-sided.
 */
export function buildRibbons(lines: readonly (readonly number[])[], o: RibbonOptions): RibbonMesh {
  if (!(o.widthM > 0) || !Number.isFinite(o.widthM)) throw new RangeError(`widthM must be > 0, got ${o.widthM}`);
  const half = o.widthM / 2;
  const positions: number[] = [];
  const indices: number[] = [];

  for (const line of lines) {
    const draped = drapePolyline(line as number[], o.heightAt, o.stepM);
    const xs: number[] = [];
    const zs: number[] = [];
    for (let i = 0; i + 2 < draped.length; i += 3) {
      // Drop a point that repeats the previous one: it has no direction.
      if (xs.length > 0 && draped[i] === xs[xs.length - 1] && draped[i + 2] === zs[zs.length - 1]) continue;
      xs.push(draped[i]);
      zs.push(draped[i + 2]);
    }
    if (xs.length < 2) continue;

    const base = positions.length / 3;
    for (let i = 0; i < xs.length; i++) {
      const prev = Math.max(0, i - 1);
      const next = Math.min(xs.length - 1, i + 1);
      let tx = xs[next] - xs[prev];
      let tz = zs[next] - zs[prev];
      const len = Math.hypot(tx, tz) || 1;
      tx /= len;
      tz /= len;
      const nx = -tz * half;
      const nz = tx * half;
      for (const side of [1, -1]) {
        const x = xs[i] + nx * side;
        const z = zs[i] + nz * side;
        positions.push(x, (Math.max(o.heightAt(x, z), o.waterLevelM) + o.liftM) * o.exaggeration, z);
      }
    }
    for (let i = 0; i + 1 < xs.length; i++) {
      const a = base + 2 * i;
      const b = a + 1;
      const c = a + 2;
      const d = a + 3;
      indices.push(a, b, c, b, d, c);
    }
  }
  return { positions: new Float32Array(positions), indices: new Uint32Array(indices) };
}
```

```ts
import type { HeightAt } from "../overlay/drape";
import {
  CREEK_COLOR,
  CREEK_LIFT_M,
  CREEK_OPACITY,
  CREEK_RENDER_ORDER,
  CREEK_STEP_M,
  CREEK_WIDTH_M,
  MODELLED_COLOR,
  MODELLED_OPACITY,
  MODELLED_WIDTH_M,
} from "./colors";
import { buildRibbons, type RibbonMesh } from "./ribbon";
import type { Creeks } from "./types";

export interface CreeksLayerOptions {
  waterLevelM: number;
  exaggeration: number;
}

function geometry(mesh: RibbonMesh): THREE.BufferGeometry {
  const g = new THREE.BufferGeometry();
  g.setAttribute("position", new THREE.BufferAttribute(mesh.positions, 3));
  g.setIndex(new THREE.BufferAttribute(mesh.indices, 1));
  g.computeBoundingSphere();
  return g;
}

function material(color: number, opacity: number): THREE.MeshBasicMaterial {
  return new THREE.MeshBasicMaterial({
    color,
    opacity,
    transparent: true,
    depthWrite: false,
    // Pull toward the camera so the draped ribbon never z-fights with the terrain beneath.
    polygonOffset: true,
    polygonOffsetFactor: -2,
    polygonOffsetUnits: -2,
    side: THREE.DoubleSide,
  });
}

/** The modelled width band of an order: 3, 4, or 5 for everything from 5 up. */
function band(order: number): 3 | 4 | 5 {
  return order >= 5 ? 5 : order >= 4 ? 4 : 3;
}

/**
 * The creek ribbons: one mesh for the hand-mapped creeks and one per width band for the modelled
 * streams (3 meshes), each toggled with its layer. Static geometry built once; nothing runs per frame.
 */
export class CreeksLayer extends THREE.Group {
  private readonly mappedGroup = new THREE.Group();
  private readonly modelledGroup = new THREE.Group();

  constructor(creeks: Creeks, heightAt: HeightAt, opts: CreeksLayerOptions) {
    super();
    this.name = "creeks";
    this.mappedGroup.name = "creeks-mapped";
    this.modelledGroup.name = "creeks-modelled";
    const common = { heightAt, waterLevelM: opts.waterLevelM, stepM: CREEK_STEP_M, liftM: CREEK_LIFT_M, exaggeration: opts.exaggeration };

    if (creeks.mapped.length > 0) {
      const mesh = new THREE.Mesh(
        geometry(buildRibbons(creeks.mapped.map((c) => c.points), { ...common, widthM: CREEK_WIDTH_M })),
        material(CREEK_COLOR, CREEK_OPACITY),
      );
      mesh.name = "creeks-mapped-ribbons";
      mesh.renderOrder = CREEK_RENDER_ORDER;
      this.mappedGroup.add(mesh);
    }
    for (const b of [3, 4, 5] as const) {
      const lines = creeks.modelled.filter((m) => band(m.order) === b).map((m) => m.points);
      if (lines.length === 0) continue;
      const mesh = new THREE.Mesh(
        geometry(buildRibbons(lines, { ...common, widthM: MODELLED_WIDTH_M[b] })),
        material(MODELLED_COLOR, MODELLED_OPACITY),
      );
      mesh.name = `creeks-modelled-order-${b}`;
      mesh.renderOrder = CREEK_RENDER_ORDER;
      this.modelledGroup.add(mesh);
    }
    this.add(this.mappedGroup, this.modelledGroup);
  }

  setMappedEnabled(on: boolean): void {
    this.mappedGroup.visible = on;
  }

  setModelledEnabled(on: boolean): void {
    this.modelledGroup.visible = on;
  }

  dispose(): void {
    this.traverse((o) => {
      if (o instanceof THREE.Mesh) {
        o.geometry.dispose();
        (o.material as THREE.Material).dispose();
      }
    });
    this.removeFromParent();
  }
}
```

```ts
import type { AppContext } from "./appContext";
import { CreeksLayer } from "./creeks/layer";
import { loadCreeks } from "./creeks/parse";
import type { HeightmapMeta } from "./heightmap";
import type { AppModule } from "./modules";
import { makeHeightSampler } from "./overlay/heightAt";

export interface CreekLayerDef {
  id: string;
  label: string;
  defaultOn: boolean;
}

/** The two creek layers. Ids are what the URL hash `l=` carries, so they must not change. */
export const CREEK_LAYERS = {
  creeks: { id: "creeks", label: "Historical creeks", defaultOn: true },
  modelled: { id: "modelled-streams", label: "Modelled streams", defaultOn: true },
} as const satisfies Record<string, CreekLayerDef>;

const count = (n: unknown): number => (typeof n === "number" && Number.isFinite(n) && n > 0 ? n : 0);

/** The toggles a dataset gets: one per layer that has lines (read defensively, meta is never validated). */
export function creekLayerDefs(meta: Pick<HeightmapMeta, "creeks_mapped" | "creeks_modelled">): CreekLayerDef[] {
  const out: CreekLayerDef[] = [];
  if (count(meta.creeks_mapped) > 0) out.push(CREEK_LAYERS.creeks);
  if (count(meta.creeks_modelled) > 0) out.push(CREEK_LAYERS.modelled);
  return out;
}

/**
 * Historical creeks: SFEI's hand-mapped creek lines and its modelled stream channels, drawn as
 * ribbons on the terrain, each with its own layer toggle. A dataset without `creeks_file`, or a
 * failed load, leaves the viewer exactly as it was (one console warning).
 */
export const creeksModule: AppModule = async (ctx: AppContext) => {
  if (!ctx.meta.creeks_file) return;
  let creeks;
  try {
    creeks = await loadCreeks(ctx.assetBase, ctx.meta);
  } catch (err) {
    console.warn("Creeks not loaded; continuing without them.", err);
    return;
  }
  if (!creeks || (creeks.mapped.length === 0 && creeks.modelled.length === 0)) return;

  const heightAt = makeHeightSampler(ctx.heights, ctx.meta);
  const layer = new CreeksLayer(creeks, heightAt, {
    waterLevelM: ctx.meta.water_level_m,
    exaggeration: ctx.verticalExaggeration,
  });
  ctx.scene.add(layer);

  const defs = new Map<string, CreekLayerDef>();
  if (creeks.mapped.length > 0) defs.set(CREEK_LAYERS.creeks.id, CREEK_LAYERS.creeks);
  if (creeks.modelled.length > 0) defs.set(CREEK_LAYERS.modelled.id, CREEK_LAYERS.modelled);

  const handles = new Map<string, ReturnType<typeof ctx.shell.addToggle>>();
  const sync = () => {
    layer.setMappedEnabled(ctx.layers.has(CREEK_LAYERS.creeks.id));
    layer.setModelledEnabled(ctx.layers.has(CREEK_LAYERS.modelled.id));
    for (const id of defs.keys()) handles.get(id)?.set(ctx.layers.has(id));
    ctx.requestRender();
  };
  for (const def of defs.values()) {
    const on = ctx.layers.register(def.id, def.defaultOn);
    handles.set(def.id, ctx.shell.addToggle(def.id, def.label, on, (v) => ctx.layers.set(def.id, v)));
  }
  ctx.layers.onChange(sync);
  sync();
};
```
